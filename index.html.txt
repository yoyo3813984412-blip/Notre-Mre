<script>

/* =================================

SET YOUR REVEAL TIME HERE

Example:

September 6, 2026
7:30 PM

================================= */

const revealTime = new Date(
"September 6, 2026 19:30:00"
).getTime();



const countdown = document.getElementById("countdown");

const countdownPage =
document.getElementById("countdownPage");

const menu =
document.getElementById("menu");


function updateCountdown() {

    const now = new Date().getTime();

    const distance =
    revealTime - now;


    if (distance <= 0) {

        countdownPage.style.display =
        "none";

        menu.style.display =
        "block";

        return;

    }


    const hours =
    Math.floor(
    distance /
    (1000 * 60 * 60)
    );


    const minutes =
    Math.floor(
    (distance %
    (1000 * 60 * 60)) /

    (1000 * 60)
    );


    const seconds =
    Math.floor(
    (distance %
    (1000 * 60)) /

    1000
    );


    countdown.innerHTML =

    String(hours)
    .padStart(2, "0")

    + " : " +

    String(minutes)
    .padStart(2, "0")

    + " : " +

    String(seconds)
    .padStart(2, "0");

}


/* Update immediately */

updateCountdown();


/* Update every second */

setInterval(
updateCountdown,
1000
);

</script>
